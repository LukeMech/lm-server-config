# One unit for the whole service: `systemctl restart disks-pod` (or
# `lm-server restart disks`) stops/starts every container below together.
# Containers in a pod share localhost, so they talk to each other on 127.0.0.1.
[Unit]
Description=disks: FileBrowser Quantum + Syncthing
ConditionPathExists=/var/lib/lm-server/env/disks/enabled
# Wait for the data disk if one is mounted there (Cockpit > Storage).
RequiresMountsFor=/var/mnt/data

[Pod]
PodName=disks
Network=disks.network
PublishPort=127.0.0.1:8001:80
PublishPort=127.0.0.1:8384:8384
PublishPort=22000:22000/tcp
PublishPort=22000:22000/udp
PublishPort=21027:21027/udp
