# One unit for the whole service: `systemctl restart remote-pod` (or
# `lm-server restart remote`) stops/starts every container below together.
# Containers in a pod share localhost, so they talk to each other on 127.0.0.1.
[Unit]
Description=remote: Apache Guacamole
ConditionPathExists=/var/lib/lm-server/env/remote/enabled
# Its data (bind-mounted from the `storage` disk, if one is set).
RequiresMountsFor=/var/lib/lm-server/volumes/remote

[Pod]
PodName=remote
Network=remote.network
PublishPort=127.0.0.1:8443:8080

[Service]
Slice=lm-server-remote.slice
