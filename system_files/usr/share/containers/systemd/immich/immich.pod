# lm-server: description Immich
# lm-server: route immich.lukemech.org http://localhost:2283
# One unit for the whole service: `systemctl restart immich-pod` (or
# `lm-server services restart immich`) stops/starts every container below together.
# Containers in a pod share localhost, so they talk to each other on 127.0.0.1.
[Unit]
Description=immich: Immich
ConditionPathExists=/var/lib/lm-server/env/immich/enabled
# Its data (bind-mounted from the `storage` disk, if one is set).
RequiresMountsFor=/var/lib/lm-server/volumes/immich

[Pod]
PodName=immich
Network=immich.network
# On the LAN too (http://<server>:2283 -- the mobile app at home, no detour
# through the tunnel), not only for cloudflared on localhost. Opened in the
# firewall as the service "immich" (/usr/lib/firewalld/services/immich.xml).
PublishPort=2283:2283

[Service]
Slice=lm-server-immich.slice
